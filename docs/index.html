Quick-Start Code Snippets

Drop-in production patterns designed for high throughput and zero boilerplate.

package main

import (
	"context"
	"fmt"
	"time"

	"github.com/umesh0492/go-libs/retry"
)

func main() {
	ctx := context.Background()

	// Declarative exponential backoff with full jitter to avoid thundering herds
	cfg := retry.Config{
		Attempts:    3,
		InitialWait: 100 * time.Millisecond,
		MaxWait:     2 * time.Second,
		Strategy:    retry.ExponentialJitter,
	}

	result, err := retry.DoWithResult(ctx, cfg, func(ctx context.Context) (string, error) {
		// Call downstream service or database query
		return "data from upstream", nil
	})

	if err != nil {
		fmt.Printf("Execution failed: %v\n", err)
		return
	}
	fmt.Printf("Success: %s\n", result)
}
package main

import (
	"context"
	"fmt"
	"time"

	"github.com/umesh0492/go-libs/circuitbreaker"
)

func main() {
	ctx := context.Background()

	// Trips open after 3 consecutive failures; stays open for 500ms before half-open probe
	cb := circuitbreaker.NewConsecutiveBreaker(3, 500*time.Millisecond)

	err := cb.Execute(ctx, func() error {
		// Outbound call to external service
		return nil
	})

	if err != nil {
		if err == circuitbreaker.ErrOpenState {
			fmt.Println("Fast fail: Circuit breaker is OPEN")
		}
		return
	}
	fmt.Println("Outbound call succeeded")
}
package main

import (
	"fmt"
	"time"

	"github.com/umesh0492/go-libs/ratelimit"
)

func main() {
	// Zero-allocation Token Bucket: 100 requests per minute capacity
	limiter := ratelimit.NewTokenBucket(100, time.Minute)

	clientIP := "192.168.1.100"
	if !limiter.Allow(clientIP) {
		fmt.Println("429 Too Many Requests: Rate limit exceeded")
		return
	}

	fmt.Println("Request allowed through rate limiter")
}
package main

import (
	"github.com/gin-gonic/gin"
	"github.com/umesh0492/go-libs/ginmw"
)

func setupEngine() *gin.Engine {
	r := gin.New()

	// Hardened zero-leak middleware stack
	r.Use(ginmw.LimitBodyDefault())            // 2MB payload ceiling
	r.Use(ginmw.Recovery())                    // Structured JSON panic recovery
	r.Use(ginmw.SecurityHeaders("my-service")) // OWASP defensive HTTP headers
	r.Use(ginmw.RequestID())                  // W3C trace correlation propagation
	r.Use(ginmw.CORS([]string{"*.example.com"}))
	r.Use(ginmw.Logger())                     // Contextual slog access logging
	r.Use(ginmw.GlobalRateLimit())            // 200 req/min IP flood barrier

	r.GET("/healthz", func(c *gin.Context) {
		c.JSON(200, gin.H{"status": "ok"})
	})

	return r
}
package main

import (
	"context"
	"log"
	"time"

	"github.com/umesh0492/go-libs/shutdown"
)

func main() {
	// Graceful shutdown manager with 15-second teardown timeout
	mgr := shutdown.New(15 * time.Second)

	// Register teardown hooks in reverse dependency order
	mgr.Register("http-server", func(ctx context.Context) error {
		log.Println("Stopping HTTP server listener...")
		return nil
	})

	mgr.Register("database-pool", func(ctx context.Context) error {
		log.Println("Closing database connections...")
		return nil
	})

	// Blocks until SIGINT/SIGTERM is caught, then executes hooks concurrently
	mgr.Wait()
}
package main

import (
	"context"
	"fmt"
	"time"

	"github.com/umesh0492/go-libs/cache"
)

type User struct {
	ID   string
	Name string
}

func main() {
	ctx := context.Background()

	// Generic type-safe cache with singleflight thundering-herd suppression
	c := cache.NewTypedCache[User](
		cache.WithCapacity(10000),
		cache.WithEvictionPolicy(cache.LRU),
	)

	user, err := c.GetOrFetch(ctx, "user:101", 5*time.Minute, func(ctx context.Context) (User, error) {
		// Only one concurrent fetcher will execute this fetch block; others await result
		return User{ID: "101", Name: "Ada Lovelace"}, nil
	})

	if err != nil {
		fmt.Printf("Cache fetch failed: %v\n", err)
		return
	}
	fmt.Printf("Retrieved user: %+v\n", user)
}
